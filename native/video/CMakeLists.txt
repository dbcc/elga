cmake_minimum_required(VERSION 3.24)
project(ElgaVideo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/EHsc /W4)
set(RTX_VIDEO_SDK "" CACHE PATH "NVIDIA RTX Video SDK 1.1.0 root")
set(OPTICAL_FLOW_SDK "" CACHE PATH "NVIDIA Optical Flow SDK 5.0.7 root")
set(ELGA_NGX_APP_ID "0" CACHE STRING "NVIDIA registered application ID; 0 for development")
option(ELGA_VIDEO_TESTS "Build deterministic and synthetic GPU tests" ON)

add_library(elga-video SHARED backend.cpp provider_nvidia.cpp)
target_compile_definitions(elga-video PRIVATE ELGA_NGX_APP_ID=${ELGA_NGX_APP_ID})
target_link_libraries(elga-video PRIVATE d3d11 dxgi d3dcompiler ole32 wintrust crypt32)
if(RTX_VIDEO_SDK)
    find_path(VSR_INCLUDE nvsdk_ngx_defs_vsr.h PATHS "${RTX_VIDEO_SDK}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(VSR_LIB nvsdk_ngx_s PATHS "${RTX_VIDEO_SDK}/lib/Windows/x64" NO_DEFAULT_PATH REQUIRED)
    target_include_directories(elga-video PRIVATE "${VSR_INCLUDE}")
    target_link_libraries(elga-video PRIVATE "${VSR_LIB}")
    target_compile_definitions(elga-video PRIVATE ELGA_WITH_VSR=1)
endif()
if(OPTICAL_FLOW_SDK)
    file(GLOB_RECURSE FRUC_HEADERS "${OPTICAL_FLOW_SDK}/NvOFFRUC.h")
    list(LENGTH FRUC_HEADERS FRUC_HEADER_COUNT)
    if(NOT FRUC_HEADER_COUNT EQUAL 1)
        message(FATAL_ERROR "Expected exactly one NvOFFRUC.h in Optical Flow SDK 5.0.7")
    endif()
    list(GET FRUC_HEADERS 0 FRUC_HEADER)
    get_filename_component(FRUC_INCLUDE "${FRUC_HEADER}" DIRECTORY)
    target_include_directories(elga-video PRIVATE "${FRUC_INCLUDE}")
    target_compile_definitions(elga-video PRIVATE ELGA_WITH_FRUC=1)
endif()
if(ELGA_VIDEO_TESTS)
    enable_testing()
    add_executable(video-timing-test timing_test.cpp)
    add_test(NAME video-timing COMMAND video-timing-test)
    add_library(elga-video-test SHARED backend.cpp provider_test.cpp)
    target_link_libraries(elga-video-test PRIVATE d3d11 dxgi d3dcompiler ole32)
    add_executable(video-backend-test backend_test.cpp)
    target_link_libraries(video-backend-test PRIVATE d3d11 dxgi)
    add_test(NAME video-backend COMMAND video-backend-test "$<TARGET_FILE:elga-video-test>")
    # Run explicitly against a packaged production DLL; never a synthetic provider.
    add_executable(video-nvidia-test nvidia_test.cpp)
    target_link_libraries(video-nvidia-test PRIVATE d3d11 dxgi)
endif()
